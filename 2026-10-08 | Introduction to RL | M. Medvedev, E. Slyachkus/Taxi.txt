import random
import time
import gymnasium as gym
import numpy as np

# 1. Инициализация окружения Taxi
# Для обучения отключаем рендеринг ради скорости.
env = gym.make("Taxi-v4")

# 2. Настройка гиперпараметров
ALPHA = 0.1  # Скорость обучения
GAMMA = 0.6  # Фактор дисконтирования (для Такси обычно берут меньше, чтобы ценились быстрые шаги)
EPSILON = 1.0  # Вероятность случайного шага
EPS_DECAY = 0.001  # Скорость снижения EPSILON
MIN_EPSILON = 0.01
EPISODES = 2000

# 3. Создание Q-таблицы
# В Taxi-v4 всего 500 состояний и 6 возможных действий
q_table = np.zeros([env.observation_space.n, env.action_space.n])

print("Начало обучения такси...")

# 4. Цикл обучения
for episode in range(EPISODES):
    state, _ = env.reset()
    done = False

    while not done:
        # Эпсилон-жадная стратегия
        if random.uniform(0, 1) < EPSILON:
            action = env.action_space.sample()  # Исследование
        else:
            action = np.argmax(q_table[state])  # Эксплуатация

        # Шаг в окружении
        next_state, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated

        # Обновление Q-значения по формуле Беллмана
        old_value = q_table[state, action]
        next_max = np.max(q_table[next_state])

        q_table[state, action] = old_value + ALPHA * (
            reward + GAMMA * next_max - old_value
        )

        state = next_state

    # Снижение EPSILON
    EPSILON = max(MIN_EPSILON, EPSILON - EPS_DECAY)

print("Обучение завершено!\n")


# 5. Демонстрация работы обученного такси
print("Запуск демонстрации (5 эпизодов)...")
# Пересоздаем окружение с включенной графикой в консоли
eval_env = gym.make("Taxi-v4", render_mode="human")

for episode in range(5):
    state, _ = eval_env.reset()
    done = False
    total_reward = 0

    print(f"\n--- Эпизод {episode + 1} ---")
    while not done:
        eval_env.render()  # Отрисовка такси в консоли
        time.sleep(0.3)  # Пауза, чтобы успеть рассмотреть шаги

        # Выбираем строго лучшее действие по нашей таблице
        action = np.argmax(q_table[state])

        state, reward, terminated, truncated, _ = eval_env.step(action)
        done = terminated or truncated
        total_reward += reward

    eval_env.render()
    print(f"Episode finished, total reward: {total_reward}")

eval_env.close()
